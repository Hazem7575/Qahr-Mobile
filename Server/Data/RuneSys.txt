	if (keyValue) then
		return keyValue

		return ""
	end
	
	return 0

-- 是否是至臻神纹
local function IsRunePerfect(itemtype)
    for _, value in pairs(RunePerfectData) do
        if value == itemtype then
           return true
        end
    end
    return false
end

-- 是否是互斥神纹
local function IsMutexRune(itemtype1, itemtype2)
    if RunePerfectSyntheticData[itemtype1] ~= nil then
       if RunePerfectSyntheticData[itemtype1][0] == itemtype2 then
            return true
       elseif RunePerfectSyntheticData[itemtype1][1] == itemtype2 then
            return true
       end
    end
    if RunePerfectSyntheticData[itemtype2] ~= nil then
        if RunePerfectSyntheticData[itemtype2][0] == itemtype1 then
             return true
        elseif RunePerfectSyntheticData[itemtype2][1] == itemtype1 then
             return true
        end
    end

    return false
end

-- 获取化臻神纹属性
local function GetRuneHuazhenAttr(key, num, attrType)
	if num == 0 then
		return 0
	end

	-- 单条
	if key == "One" then
		-- 这里num是具体的化臻位置
		local attr = RuneHuazhenAttr[num]
		if attr == nil then
			return ""
		end

		local strTemp = ""
		local strFormat = ""
		local strText = ""
		for k, v in pairs(attr) do
			strTemp = string.format("STR_HUAZHEN_ATTR_NEW_TYPE_%d", k)
			strTemp = g_objLuaUIMgr:GetCnRes(strTemp)
			if (k >= RUNEATTR.USER_ATTRIBUTE_PHYSIC_CRIT_RATE and k <= RUNEATTR.USER_ATTRIBUTE_FIRM_DEFENSE_RATE) or k == RUNEATTR.USER_ATTRIBUTE_TYPE_HEAVY_HIT or k == RUNEATTR.USER_ATTRIBUTE_TYPE_DEFENSE_HEAVY_HIT then
				strFormat = string.format(strTemp, v / 100.0)
			else
				strFormat = string.format(strTemp, v)
			end
			
			strText = strText..strFormat
		end

		if strText ~= "" then
			strText = g_objLuaUIMgr:GetCnRes("STR_HUAZHEN_GONGMING") ..strText
			if attrType then
				strText = strText..g_objLuaUIMgr:GetCnRes("STR_HUAZHEN_GONGMING_UNLOCK")
			end
		end

		return LuaUIMgr_StringCodePage(strText)
	end

	if key == "Life" then
		local value = 0
		for i = 1, num, 1 do 
			if RuneHuazhenAttr[i] == nil then
				break
			end
	
			local attrValue = RuneHuazhenAttr[i][attrType]
			if attrValue ~= nil then
				value = value + attrValue
			end	
		end
		return value
	end

	-- 多条累加
	local value = 0
	for i = 1, num, 1 do 
		if RuneHuazhenAttr[i] == nil then
			break
		end

		local attrValue = RuneHuazhenAttr[i][attrType]
		if attrValue ~= nil then
			value = value + attrValue
		end	
	end

	if value == 0 then
		return ""
	end

	local strText = g_objLuaUIMgr:GetCnRes(string.format("STR_HUAZHEN_ATTR_TYPE_%d", attrType))
	if attrType >= RUNEATTR.USER_ATTRIBUTE_PHYSIC_CRIT_RATE and attrType <= RUNEATTR.USER_ATTRIBUTE_FIRM_DEFENSE_RATE then
		strText = string.format(strText, value / 100.0)
	else
		strText = string.format(strText, value)
	end

	return LuaUIMgr_StringCodePage(strText)
end
